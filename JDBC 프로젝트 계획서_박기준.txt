영화예매 프로그램
영화를 예매를 할려면 영화정보가 있어야하고
좌석 정보가 필요하고
좌석 정보에서 <- 이미 예매한 좌석 있을 가능성 어떻게 판단할건지 <- 예매티켓 에 좌석정보와 예매 여부가 있다 
<- 그럼 예매티켓에서의 예매 여부는 무조건 1인가?

영화정보 , 예매 티켓 , 좌석 정보 
예매를 할려면 -> 영화정보와 좌석 정보가 필요해 

좌석정보는 예매한 좌석을 알려면 -> 예매티켓의 예매 여부를 알아야해

예매를 하면 A , 1 이 들어옴 이걸 문자열로 치환 -> A열 1행 하는것보다

티켓과 예매의 행 열을 동일선상에 놓고 그럼 서로 참조인가?
둘다 필요로 하는거같은데 예매를 할려면 좌석정보가 즉 행, 열 정보가 필요하고 
좌석도 티켓의 예매여부를 알아야 좌석에서의 예매가 됬는지 안됬는지 알수있다

누구를 부모로 둬야하지
좌석은 영화관에서 고정이고 티켓은 뽑는거니까 

좌석을 부모로 하고 

primary key 하나의 테이블에 중복해서 사용못함
-> 복합 기본키로 사용해야한다고함 -> 복합 기본키 쓰지말고 unique로 쓰자 

예매 인원 1명으로 고정

---영화-------------------------

drop table movies cascade constraints;

create table movies(
	movie_id number primary key ,
	screen_id number not null , 
	title varchar2(36) not null , 
	movie_start date not null ,
	movie_time number not null 
);

-- 좌석 정보 --------------------

drop table seat_info cascade constraints;

create table seat_info (
    seat_id number primary key ,
    screen_id number not null,
    seat_row varchar2(3) not null ,
    seat_col number not null ,
	unique (screen_id, seat_row, seat_col)
);


--- 예매 정보 --------------------

drop table ticket ; 

create table ticket (
    ticket_id number primary key ,
    movie_id number not null references movies(movie_id),
    seat_id number not null references seat_info(seat_id),
    ticket_num number default 1 not null check (ticket_num = 1),
	unique (movie_id, seat_id)
);

-- 시퀀스 ----------------------

drop sequence movie_seq;
create sequence movie_seq
	increment by 1
	start with 1 
	minvalue 1
	maxvalue 10000
	nocache;


drop sequence ticket_seq;
create sequence ticket_seq
	increment by 1
	start with 1 
	minvalue 1
	maxvalue 10000
	nocache;

drop sequence seat_seq;
create sequence seat_seq
	increment by 1
	start with 1 
	minvalue 1
	maxvalue 250
	nocache;

--전체 에매 정보 조회 할때 SQL 문

select 
k.ticket_id , k.ticket_num ,
m.movie_id , m.screen_id, m.title ,m.movie_start , m.movie_time , 
s.seat_id , s.seat_row , s.seat_col , s.screen_id  
from ticket k
join movies m on k.movie_id = m.movie_id
join seat_info s on k.seat_id = s.seat_id ;
